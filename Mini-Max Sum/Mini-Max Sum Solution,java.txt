        }

                max=a[i];
            }else{
                max=max;
            }
        }
        long min=a[0];
        for(int i=0;i<5;i++){
            if(a[i]<min){
                min=a[i];
            }else{
                min=min;
            }
        }
        for(int i=0;i<5;i++){
            sum=sum+a[i];
        }
    long sum1=sum-max;
        long sum2=sum-min;
